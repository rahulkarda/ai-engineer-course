## Exercise 1: Exactly-Once Semantics with Idempotency Key Store

Deduplicate messages using a key store backed by a `dict`, with periodic TTL cleanup.

```python
import asyncio
import logging
import os
import time
from dataclasses import dataclass, field
from typing import Any
from uuid import uuid4

logger = logging.getLogger(__name__)

# idempotency key TTL in seconds; tune to message retention window
IDEMPOTENCY_TTL: float = float(os.getenv("IDEMPOTENCY_TTL_SECONDS", "60"))


@dataclass
class IdempotencyEntry:
    seen_at: float = field(default_factory=time.monotonic)


class IdempotencyStore:
    """
    Thread-safe (single-event-loop) key store with TTL eviction.
    Keys older than `ttl_seconds` are purged lazily on each `is_duplicate` check.
    """

    def __init__(self, ttl_seconds: float = IDEMPOTENCY_TTL) -> None:
        self._store: dict[str, IdempotencyEntry] = {}
        self._ttl = ttl_seconds

    def _evict_expired(self) -> None:
        cutoff = time.monotonic() - self._ttl
        expired = [k for k, v in self._store.items() if v.seen_at < cutoff]
        for k in expired:
            del self._store[k]
        if expired:
            logger.debug("evicted %d expired idempotency keys", len(expired))

    def is_duplicate(self, key: str) -> bool:
        self._evict_expired()
        if key in self._store:
            return True
        self._store[key] = IdempotencyEntry()
        return False

    @property
    def size(self) -> int:
        return len(self._store)


@dataclass
class Message:
    idempotency_key: str
    payload: Any


class ExactlyOnceBroker:
    def __init__(self, ttl_seconds: float = IDEMPOTENCY_TTL) -> None:
        self._queue: asyncio.Queue[Message] = asyncio.Queue()
        self._store = IdempotencyStore(ttl_seconds=ttl_seconds)
        self.duplicate_count: int = 0
        self.processed_count: int = 0

    async def publish(self, payload: Any, idempotency_key: str | None = None) -> str:
        key = idempotency_key or str(uuid4())
        msg = Message(idempotency_key=key, payload=payload)
        await self._queue.put(msg)
        return key

    async def consume(self) -> Any | None:
        """Returns payload if first time seeing the key, None if duplicate."""
        msg = await self._queue.get()
        self._queue.task_done()
        if self._store.is_duplicate(msg.idempotency_key):
            self.duplicate_count += 1
            logger.info("duplicate rejected key=%s", msg.idempotency_key)
            return None
        self.processed_count += 1
        return msg.payload


async def demo_exactly_once() -> None:
    broker = ExactlyOnceBroker(ttl_seconds=5.0)

    key = await broker.publish({"order": 1}, idempotency_key="order-1")
    await broker.publish({"order": 1}, idempotency_key="order-1")  # duplicate
    await broker.publish({"order": 2}, idempotency_key="order-2")

    results = [await broker.consume() for _ in range(3)]
    live = [r for r in results if r is not None]

    assert len(live) == 2, f"expected 2 unique, got {live}"
    assert broker.duplicate_count == 1
    assert broker.processed_count == 2
    logger.info(
        "exactly-once demo passed — processed=%d duplicates_rejected=%d",
        broker.processed_count, broker.duplicate_count,
    )


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    asyncio.run(demo_exactly_once())
```

**Key points:**
- Idempotency keys must be generated by the *producer* (e.g., `uuid4()`) and included in the message, not assigned by the broker.
- Lazy eviction on every `is_duplicate` call is O(n) but avoids a background task. For high-throughput paths, run `_evict_expired` on a schedule instead.
- `IDEMPOTENCY_TTL` is pulled from the environment via `os.getenv` so it's configurable without code changes.

## Exercise 2: Exponential Backoff on Retry

At-least-once broker where retry delay doubles each attempt.

```python
import asyncio
import logging
import os
import time
from dataclasses import dataclass, field
from typing import Any, Callable, Awaitable

logger = logging.getLogger(__name__)

MAX_RETRIES: int = int(os.getenv("MAX_RETRIES", "5"))
BASE_DELAY_S: float = float(os.getenv("BASE_RETRY_DELAY_SECONDS", "0.1"))
MAX_DELAY_S: float = float(os.getenv("MAX_RETRY_DELAY_SECONDS", "30.0"))


@dataclass
class RetryableMessage:
    payload: Any
    attempts: int = 0
    last_error: str | None = None


Handler = Callable[[Any], Awaitable[None]]


class AtLeastOnceBroker:
    """
    Delivers messages at least once. On handler exception, retries with
    exponential backoff (capped at MAX_DELAY_S). After MAX_RETRIES, routes
    to DLQ.
    """

    def __init__(
        self,
        max_retries: int = MAX_RETRIES,
        base_delay: float = BASE_DELAY_S,
        max_delay: float = MAX_DELAY_S,
    ) -> None:
        self._queue: asyncio.Queue[RetryableMessage] = asyncio.Queue()
        self._dlq: list[RetryableMessage] = []
        self._max_retries = max_retries
        self._base_delay = base_delay
        self._max_delay = max_delay

    async def publish(self, payload: Any) -> None:
        await self._queue.put(RetryableMessage(payload=payload))

    def _backoff_delay(self, attempt: int) -> float:
        delay = self._base_delay * (2 ** attempt)
        return min(delay, self._max_delay)

    async def consume(self, handler: Handler) -> None:
        """
        Fetch one message and call handler.
        On failure: schedule retry with backoff or route to DLQ.
        """
        msg = await self._queue.get()
        try:
            await handler(msg.payload)
            self._queue.task_done()
            logger.debug("processed after %d attempt(s)", msg.attempts + 1)
        except Exception as exc:
            msg.attempts += 1
            msg.last_error = str(exc)
            if msg.attempts >= self._max_retries:
                self._dlq.append(msg)
                self._queue.task_done()
                logger.error(
                    "DLQ: giving up after %d attempts, last_error=%s",
                    msg.attempts, msg.last_error,
                )
            else:
                delay = self._backoff_delay(msg.attempts - 1)
                logger.warning(
                    "attempt %d failed (%s), retrying in %.2fs",
                    msg.attempts, exc, delay,
                )
                self._queue.task_done()
                await asyncio.sleep(delay)
                await self._queue.put(msg)

    async def drain(self, handler: Handler) -> None:
        while not self._queue.empty():
            await self.consume(handler)

    @property
    def dlq(self) -> list[RetryableMessage]:
        return list(self._dlq)


async def demo_backoff() -> None:
    broker = AtLeastOnceBroker(max_retries=3, base_delay=0.01, max_delay=1.0)
    call_log: list[tuple[Any, float]] = []

    async def flaky_handler(payload: Any) -> None:
        call_log.append((payload, time.monotonic()))
        if len([p for p, _ in call_log if p == payload]) < 3:
            raise ValueError("transient error")

    await broker.publish({"id": 1})

    # process until either success or DLQ
    for _ in range(10):
        if broker._queue.empty() and not broker._dlq:
            break
        if not broker._queue.empty():
            await broker.consume(flaky_handler)

    successes = [p for p, _ in call_log if p == {"id": 1}]
    assert len(successes) == 3, f"expected 3 calls (2 failures + 1 success), got {len(successes)}"

    # verify delays doubled between retries
    times = [t for p, t in call_log if p == {"id": 1}]
    delays = [times[i + 1] - times[i] for i in range(len(times) - 1)]
    logger.info("retry delays (s): %s", [f"{d:.3f}" for d in delays])
    assert delays[1] > delays[0], "second delay should be longer than first"
    logger.info("exponential backoff demo passed")


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    asyncio.run(demo_backoff())
```

**Key points:**
- Delay formula: `base * 2^(attempt-1)`, capped at `max_delay`. Add jitter (`random.uniform(0.5*delay, delay)`) in production to prevent thundering herd.
- Failed messages are re-enqueued (not re-processed in a loop) so other messages are not starved during the backoff sleep.

## Exercise 3: Batch Producer with Batch-Level Retry

Produce messages in batches; retry the whole batch only if the entire batch fails.

```python
import asyncio
import logging
import os
from dataclasses import dataclass, field
from typing import Any, Callable, Awaitable

logger = logging.getLogger(__name__)

BATCH_SIZE: int = int(os.getenv("BATCH_SIZE", "10"))
MAX_BATCH_RETRIES: int = int(os.getenv("MAX_BATCH_RETRIES", "3"))


@dataclass
class Batch:
    messages: list[Any]
    attempt: int = 0
    batch_id: str = field(default_factory=lambda: str(id(object())))


BatchHandler = Callable[[list[Any]], Awaitable[None]]


class BatchProducer:
    """
    Accumulates messages into fixed-size batches, then delivers with retry.
    Retries the entire batch only if delivery raises an exception.
    """

    def __init__(
        self,
        batch_size: int = BATCH_SIZE,
        max_retries: int = MAX_BATCH_RETRIES,
        base_delay: float = 0.05,
    ) -> None:
        self._batch_size = batch_size
        self._max_retries = max_retries
        self._base_delay = base_delay
        self._buffer: list[Any] = []
        self._dlq: list[Batch] = []

    def add(self, message: Any) -> bool:
        """Add a message; returns True when a full batch is ready to flush."""
        self._buffer.append(message)
        return len(self._buffer) >= self._batch_size

    async def flush(self, handler: BatchHandler) -> None:
        """Send buffered messages as one batch, retrying on failure."""
        if not self._buffer:
            return
        batch = Batch(messages=self._buffer[:])
        self._buffer.clear()
        await self._deliver(batch, handler)

    async def _deliver(self, batch: Batch, handler: BatchHandler) -> None:
        while True:
            try:
                await handler(batch.messages)
                logger.info(
                    "batch id=%s delivered %d messages in %d attempt(s)",
                    batch.batch_id, len(batch.messages), batch.attempt + 1,
                )
                return
            except Exception as exc:
                batch.attempt += 1
                if batch.attempt >= self._max_retries:
                    self._dlq.append(batch)
                    logger.error(
                        "batch id=%s sent to DLQ after %d attempts: %s",
                        batch.batch_id, batch.attempt, exc,
                    )
                    return
                delay = self._base_delay * (2 ** (batch.attempt - 1))
                logger.warning(
                    "batch id=%s attempt %d failed (%s), retrying in %.3fs",
                    batch.batch_id, batch.attempt, exc, delay,
                )
                await asyncio.sleep(delay)

    @property
    def dlq(self) -> list[Batch]:
        return list(self._dlq)


async def demo_batch_retry() -> None:
    producer = BatchProducer(batch_size=5, max_retries=3, base_delay=0.01)
    attempt_log: list[int] = []

    async def unreliable_handler(messages: list[Any]) -> None:
        attempt_log.append(len(attempt_log) + 1)
        if attempt_log[-1] < 2:  # fail on first attempt
            raise ConnectionError("downstream unavailable")

    for i in range(5):
        producer.add({"id": i})

    await producer.flush(unreliable_handler)

    assert len(attempt_log) == 2, f"expected 2 attempts (1 fail + 1 success), got {attempt_log}"
    assert len(producer.dlq) == 0
    logger.info("batch retry demo passed — attempts=%d dlq=%d", len(attempt_log), len(producer.dlq))


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    asyncio.run(demo_batch_retry())
```

**Key points:**
- The buffer is cleared *before* delivery so a retry re-sends the original batch, not an appended one.
- Batch-level retry is appropriate when the handler is transactional (all-or-nothing). For partial failures, fall back to per-message retry on only the failed items.

## Exercise 4: Delivery Metrics

Track `delivery_attempts`, `successful_deliveries`, `duplicate_rejections`, and `dlq_count`.

```python
import asyncio
import logging
import os
from dataclasses import dataclass, field
from typing import Any, Callable, Awaitable
from uuid import uuid4

logger = logging.getLogger(__name__)


@dataclass
class DeliveryMetrics:
    delivery_attempts: int = 0
    successful_deliveries: int = 0
    duplicate_rejections: int = 0
    dlq_count: int = 0

    def summary(self) -> dict[str, int]:
        return {
            "delivery_attempts": self.delivery_attempts,
            "successful_deliveries": self.successful_deliveries,
            "duplicate_rejections": self.duplicate_rejections,
            "dlq_count": self.dlq_count,
        }


@dataclass
class TrackedMessage:
    idempotency_key: str
    payload: Any
    attempts: int = 0


Handler = Callable[[Any], Awaitable[None]]


class MetricsBroker:
    def __init__(self, max_retries: int = 3, base_delay: float = 0.05) -> None:
        self._queue: asyncio.Queue[TrackedMessage] = asyncio.Queue()
        self._seen_keys: set[str] = set()
        self._max_retries = max_retries
        self._base_delay = base_delay
        self.metrics = DeliveryMetrics()

    async def publish(self, payload: Any, idempotency_key: str | None = None) -> None:
        key = idempotency_key or str(uuid4())
        if key in self._seen_keys:
            self.metrics.duplicate_rejections += 1
            logger.debug("duplicate publish rejected key=%s", key)
            return
        self._seen_keys.add(key)
        await self._queue.put(TrackedMessage(idempotency_key=key, payload=payload))

    async def consume(self, handler: Handler) -> None:
        msg = await self._queue.get()
        self.metrics.delivery_attempts += 1
        try:
            await handler(msg.payload)
            self.metrics.successful_deliveries += 1
            self._queue.task_done()
        except Exception as exc:
            msg.attempts += 1
            if msg.attempts >= self._max_retries:
                self.metrics.dlq_count += 1
                self._queue.task_done()
                logger.error("DLQ key=%s after %d attempts: %s", msg.idempotency_key, msg.attempts, exc)
            else:
                self._queue.task_done()
                await asyncio.sleep(self._base_delay * (2 ** msg.attempts))
                await self._queue.put(msg)

    async def drain(self, handler: Handler) -> None:
        while not self._queue.empty():
            await self.consume(handler)


async def demo_metrics() -> None:
    broker = MetricsBroker(max_retries=2, base_delay=0.01)

    # one normal message
    await broker.publish({"id": 1}, idempotency_key="k1")
    # one duplicate
    await broker.publish({"id": 1}, idempotency_key="k1")
    # one message that will always fail → DLQ
    await broker.publish({"id": 2}, idempotency_key="k2")

    async def handler(payload: Any) -> None:
        if payload.get("id") == 2:
            raise RuntimeError("always fails")

    await broker.drain(handler)

    m = broker.metrics.summary()
    logger.info("metrics: %s", m)

    assert m["duplicate_rejections"] == 1
    assert m["successful_deliveries"] == 1
    assert m["dlq_count"] == 1
    assert m["delivery_attempts"] >= 3  # k1 once + k2 twice (2 retries)


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    asyncio.run(demo_metrics())
```

**Key points:**
- `delivery_attempts` counts every attempt including retries, giving you an effective delivery ratio: `successful_deliveries / delivery_attempts`.
- Expose `metrics.summary()` to your APM or Prometheus endpoint for dashboarding.

## Exercise 5: Saga Pattern — Two Consumers Must Both Ack

A message is only "done" when two designated consumers both acknowledge it.

```python
import asyncio
import logging
from collections import defaultdict
from dataclasses import dataclass, field
from typing import Any
from uuid import uuid4

logger = logging.getLogger(__name__)


@dataclass
class SagaMessage:
    message_id: str
    payload: Any


@dataclass
class SagaState:
    acks: set[str] = field(default_factory=set)
    completed: bool = False


class SagaBroker:
    """
    Fanout broker requiring acks from all registered participants before
    marking a message complete. Uncompleted messages are accessible via
    `pending`.
    """

    def __init__(self, participants: list[str]) -> None:
        self._participants = set(participants)
        # per-participant delivery queue
        self._queues: dict[str, asyncio.Queue[SagaMessage]] = {
            p: asyncio.Queue() for p in participants
        }
        # per-message ack tracking
        self._saga_states: dict[str, SagaState] = {}
        self._completed: list[str] = []
        self._lock = asyncio.Lock()

    async def publish(self, payload: Any) -> str:
        msg_id = str(uuid4())
        msg = SagaMessage(message_id=msg_id, payload=payload)
        async with self._lock:
            self._saga_states[msg_id] = SagaState()
        for q in self._queues.values():
            await q.put(msg)
        logger.debug("published msg_id=%s to %d participants", msg_id, len(self._participants))
        return msg_id

    async def consume(self, participant: str) -> SagaMessage:
        if participant not in self._queues:
            raise ValueError(f"unknown participant: {participant!r}")
        return await self._queues[participant].get()

    async def ack(self, participant: str, message_id: str) -> bool:
        """
        Record ack from participant. Returns True when all participants have acked
        (saga step complete).
        """
        if participant not in self._participants:
            raise ValueError(f"unknown participant: {participant!r}")
        async with self._lock:
            state = self._saga_states.get(message_id)
            if state is None:
                raise KeyError(f"unknown message_id: {message_id}")
            state.acks.add(participant)
            if self._participants <= state.acks and not state.completed:
                state.completed = True
                self._completed.append(message_id)
                logger.info("saga complete for msg_id=%s", message_id)
                return True
        return False

    @property
    def completed(self) -> list[str]:
        return list(self._completed)

    @property
    def pending(self) -> list[str]:
        return [
            mid for mid, s in self._saga_states.items() if not s.completed
        ]


async def saga_participant(
    name: str,
    broker: SagaBroker,
    n_messages: int,
    delay: float = 0.01,
) -> None:
    for _ in range(n_messages):
        msg = await broker.consume(name)
        await asyncio.sleep(delay)  # simulate processing
        done = await broker.ack(name, msg.message_id)
        logger.debug("%s acked msg_id=%s complete=%s", name, msg.message_id, done)


async def demo_saga() -> None:
    participants = ["inventory", "billing"]
    broker = SagaBroker(participants=participants)

    n = 5
    msg_ids = [await broker.publish({"order_id": i}) for i in range(n)]

    await asyncio.gather(
        saga_participant("inventory", broker, n, delay=0.01),
        saga_participant("billing", broker, n, delay=0.02),
    )

    assert len(broker.completed) == n, f"expected {n} completed sagas, got {len(broker.completed)}"
    assert set(broker.completed) == set(msg_ids)
    assert broker.pending == []
    logger.info("saga demo passed — %d/%d sagas completed", len(broker.completed), n)


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    asyncio.run(demo_saga())
```

**Key points:**
- Each participant gets its own queue (fan-out on publish), so both see every message independently.
- `SagaState.acks` is a set; completion is triggered the first time `participants <= acks` becomes true.
- The `asyncio.Lock` around `_saga_states` prevents a race where both participants ack simultaneously and both see `completed = False` before either sets it.
- Real sagas also need compensating transactions on partial failure — this implementation covers the happy path; add a `nack` method and rollback callbacks for production use.
