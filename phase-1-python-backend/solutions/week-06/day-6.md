## Part 2 — Alembic migration (key additions)

The autogenerated migration may miss PostgreSQL-specific index options. A complete `upgrade()` includes:

```python
import sqlalchemy as sa
from alembic import op


def upgrade() -> None:
    # Tables created by autogenerate ...

    # Custom indexes
    op.create_index(
        "idx_customers_active",
        "customers",
        ["is_active"],
        postgresql_where=sa.text("is_active = true"),
    )
    op.create_index("idx_accounts_customer_id", "accounts", ["customer_id"])
    op.create_index(
        "idx_transactions_created_at",
        "transactions",
        [sa.text("created_at DESC")],
    )
    op.create_index(
        "idx_transactions_account_created",
        "transactions",
        ["account_id", sa.text("created_at DESC")],
    )
    op.create_index(
        "idx_transactions_pending",
        "transactions",
        ["account_id", "created_at"],
        postgresql_where=sa.text("status = 'pending'"),
    )
    op.create_index(
        "idx_transactions_report",
        "transactions",
        ["account_id", sa.text("created_at DESC")],
        postgresql_include=["amount", "direction", "category_id"],
    )
    op.create_index(
        "idx_audit_table_record",
        "audit_logs",
        ["table_name", "record_id", sa.text("changed_at DESC")],
    )


def downgrade() -> None:
    op.drop_index("idx_audit_table_record", table_name="audit_logs")
    op.drop_index("idx_transactions_report", table_name="transactions")
    op.drop_index("idx_transactions_pending", table_name="transactions")
    op.drop_index("idx_transactions_account_created", table_name="transactions")
    op.drop_index("idx_transactions_created_at", table_name="transactions")
    op.drop_index("idx_accounts_customer_id", table_name="accounts")
    op.drop_index("idx_customers_active", table_name="customers")
    # Tables dropped by autogenerate downgrade ...
```

## Part 5 — AuditLog model and SQLAlchemy event listener

```python
# src/financial_app/models.py — add AuditLog model
from sqlalchemy import BigInteger, DateTime, Index, Text, func
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column


class AuditLog(Base):
    __tablename__ = "audit_logs"
    __table_args__ = (
        Index("idx_audit_table_record", "table_name", "record_id"),
    )

    log_id:     Mapped[int]            = mapped_column(BigInteger, primary_key=True)
    table_name: Mapped[str]            = mapped_column(Text, nullable=False)
    record_id:  Mapped[int]            = mapped_column(BigInteger, nullable=False)
    action:     Mapped[str]            = mapped_column(Text, nullable=False)
    old_data:   Mapped[dict | None]    = mapped_column(JSONB)
    new_data:   Mapped[dict | None]    = mapped_column(JSONB)
    changed_by: Mapped[str | None]     = mapped_column(Text)
    changed_at: Mapped[object]         = mapped_column(
        DateTime(timezone=True), server_default=func.now(), nullable=False
    )
```

```python
# src/financial_app/audit.py
import logging
from sqlalchemy import event, inspect
from sqlalchemy.orm import Session
from .models import Account, Transaction, AuditLog

logger = logging.getLogger(__name__)

AUDITED_MODELS = (Account, Transaction)


def _to_dict(instance) -> dict:
    """Serialize SQLAlchemy instance to plain dict (scalar columns only)."""
    insp = inspect(instance)
    return {
        c.key: getattr(instance, c.key)
        for c in insp.mapper.column_attrs
    }


@event.listens_for(Session, "after_flush")
def _after_flush(session: Session, flush_context) -> None:
    for instance in session.new:
        if isinstance(instance, AUDITED_MODELS):
            _insert_audit(session, instance, "INSERT", old_data=None)

    for instance in session.dirty:
        if isinstance(instance, AUDITED_MODELS):
            # Capture the state before the flush using history
            _insert_audit(session, instance, "UPDATE", old_data=None)

    for instance in session.deleted:
        if isinstance(instance, AUDITED_MODELS):
            _insert_audit(session, instance, "DELETE", old_data=_to_dict(instance))


def _insert_audit(session: Session, instance, action: str, old_data: dict | None) -> None:
    table_name = instance.__tablename__
    # PK column name by convention
    pk_col = inspect(type(instance)).mapper.primary_key[0].key
    record_id = getattr(instance, pk_col, None)
    if record_id is None:
        logger.warning("Audit: could not determine PK for %s", table_name)
        return

    log = AuditLog(
        table_name=table_name,
        record_id=record_id,
        action=action,
        old_data=old_data,
        new_data=_to_dict(instance) if action != "DELETE" else None,
    )
    session.add(log)
```
